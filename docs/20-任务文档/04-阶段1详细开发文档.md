// ========== 总监对话模式 - 消息store ==========
const directorStores = /* @__PURE__ */ new Map();

function createDirectorStore(sessionId) {
    let state = {
        messages: [],       // {id, role, content, parsed?, ts}
        status: "idle",     // idle | processing | done | error
        config: {
            autoForward: true  // 自动转发到对话
        }
    };
    const listeners = /* @__PURE__ */ new Set();

    function notify() {
        for (const fn of listeners) fn(state);
    }

    return {
        sessionId,
        getState: () => state,
        subscribe: (fn) => {
            listeners.add(fn);
            return () => listeners.delete(fn);
        },
        addMessage: (msg) => {
            state = {
                ...state,
                messages: [...state.messages, { id: `dm-${Date.now()}-${Math.random().toString(36).slice(2,6)}`, ts: Date.now(), ...msg }]
            };
            notify();
        },
        setStatus: (status) => {
            state = { ...state, status };
            notify();
        },
        setConfig: (patch) => {
            state = { ...state, config: { ...state.config, ...patch } };
            notify();
        },
        reset: () => {
            state = { messages: [], status: "idle", config: { autoForward: true } };
            notify();
        }
    };
}

function directorStoreFactory(sessionId) {
    if (!directorStores.has(sessionId)) {
        directorStores.set(sessionId, createDirectorStore(sessionId));
    }
    return directorStores.get(sessionId);
}

function useDirectorStore(store) {
    // 使用React的useSyncExternalStore模式（与原代码一致，通过react命名空间调用）
    const subscribe = (0, react.useCallback)((fn) => store.subscribe(fn), [store]);
    return (0, react.useSyncExternalStore)(subscribe, store.getState, store.getState);
}

### 注意（重要：格式转换）

本文件代码中使用了简写的 `useState`/`useEffect`/`jsx` 等，**实施时必须转换为原文件的调用格式**：

| 简写 | 原文件格式 |
|------|-----------|
| `useState(x)` | `(0, react.useState)(x)` |
| `useEffect(fn, deps)` | `(0, react.useEffect)(fn, deps)` |
| `useCallback(fn, deps)` | `(0, react.useCallback)(fn, deps)` |
| `useRef(x)` | `(0, react.useRef)(x)` |
| `useMemo(fn, deps)` | `(0, react.useMemo)(fn, deps)` |
| `jsx(type, props)` | `(0, react_jsx_runtime.jsx)(type, props)` |
| `jsxs(type, props)` | `(0, react_jsx_runtime.jsxs)(type, props)` |

**依据**：原文件第10-11行 `let react_jsx_runtime = require("react/jsx-runtime"); let react = require("react");`，第5383-5388行使用 `(0, react.useMemo)()`、`(0, react.useRef)()`、`(0, react.useState)()` 格式。

**M3中需转换的调用清单**：
- `useState("")` ×2（inputText, settingsOpen）
- `useRef(null)` ×1（messagesEndRef）
- `useEffect(...)` ×1（自动滚动）
- `useCallback(...)` ×3（handleSubmit, handleConfirmSend, toggleAutoForward）
- `jsx(...)` 全部
- `jsxs(...)` 全部

**M2中需转换的调用清单**：
- `useCallback(...)` ×1（useDirectorStore中的subscribe）
- `useSyncExternalStore(...)` ×1 → `(0, react.useSyncExternalStore)(...)`

// ========== 总监对话模式 - mock处理 + 转发 ==========

/**
 * Mock总监处理（阶段1用，阶段2替换为真实Ollama调用）
 * 模拟1s延迟后返回固定格式的整理结果
 * @param {string} sessionId - 会话ID
 * @param {string} userText - 用户输入
 * @param {object} store - 总监store
 * @param {function} t - locale函数
 * @param {function} onForward - 转发回调(instruction)，autoForward时调用
 */
function directorProcessMock(sessionId, userText, store, t, onForward) {
    // 1. 添加用户消息
    store.addMessage({ role: "user", content: userText });
    store.setStatus("processing");

    // 2. 模拟处理延迟
    setTimeout(() => {
        try {
            // 3. 构造mock结果
            const mockParsed = {
                instruction: `【总监整理】${userText}`,
                model: "deepseek-chat",
                branch: "沿用当前分支",
                reasoning: `Mock推理：用户输入"${userText}"，判定为日常对话任务，建议使用deepseek-chat模型，沿用当前分支上下文。`
            };

            // 4. 添加总监回复
            store.addMessage({
                role: "assistant",
                content: `${t("director.mockResult")}\n\n${mockParsed.reasoning}`,
                parsed: mockParsed
            });
            store.setStatus("done");

            // 5. 自动转发（如果开启）
            if (store.getState().config.autoForward && typeof onForward === "function") {
                onForward(mockParsed.instruction);
            }
        } catch (err) {
            store.addMessage({ role: "system", content: `总监处理失败: ${err.message}` });
            store.setStatus("error");
        }
    }, 1000);
}
// ========== mock处理结束 ==========
```

// ========== 总监对话模式 - tab注册 ==========
slots.register({
    name: "conversation.view",
    id: "director",
    order: -1,  // 排最前（对话是0，轨迹是10）
    label: () => t("view.director"),
    locale: NS,
    store: chatStore,  // 复用chatStore，保持与对话/轨迹的视图切换一致
    inject: (sessionId, actions) => {
        // 获取conversation service用于转发
        const conversation = concreteConversation(ctx);
        const scoped = scopedConversation(sessions, sessionId);

        return {
            // 转发到原对话系统（DirectorView通过props.sendToChat调用）
            sendToChat: (text) => {
                const session = sessions.list.getSnapshot().byId[sessionId];
                if (session) {
                    conversation.sendSession(session, text, [], "chat").catch((err) => {
                        console.error("[Director] sendSession failed:", err);
                    });
                }
            },
            // 复用原对话的方法
            loadOlder: () => scoped.loadOlder(),
            loadImage: (attachment) => conversation.resolveImage(sessionId, attachment)
        };
    }
}, DirectorView);
// ========== 总监tab注册结束 ==========
```

### 关键说明
1. **order: -1**：确保总监tab排在对话(0)和轨迹(10)之前
2. **store: chatStore**：复用原chatStore，这样 `actions.setView("director")` 可以正常切换视图
3. **inject返回sendToChat**：DirectorView通过这个方法转发到原对话
4. **sendToChat实现**：通过 `sessions.list.getSnapshot().byId[sessionId]` 获取session对象，再调用 `conversation.sendSession(session, text, [], "chat")`
5. **concreteConversation和scopedConversation**：这两个函数在apply()函数内已定义（对话tab注册中使用了），可直接复用
